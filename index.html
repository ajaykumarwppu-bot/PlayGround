const PaperCut = () => {
  const f = useCurrentFrame();
  const sunY = interpolate(f, [0,300], [-40, 60]);
  return (<AbsoluteFill style={{background:"#bfe3ec"}}>
    <Cloud x={interpolate(f,[0,300],[0,400])} />
    <Mountains parallax={f*0.4} />
    <Title opacity={interpolate(f,[0,30],[0,1])} />

};
